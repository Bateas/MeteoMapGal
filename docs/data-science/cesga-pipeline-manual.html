FASE 2: Preparar TimescaleDB

2.1 Ampliar disco LXC 306 en Proxmox

Proxmox UI → LXC 306 → Hardware → Root Disk → Resize → +30GB

No hace falta reiniciar.

2.2 Crear tablas 
sudo -u postgres psql -d meteomapgal

-- WRF hindcast (viento, temp, precip...)
CREATE TABLE IF NOT EXISTS wrf_hindcast (
    time TIMESTAMPTZ NOT NULL, spot_id TEXT NOT NULL,
    lat REAL, lon REAL, wind_speed REAL, wind_dir REAL,
    wind_gust REAL, temp REAL, humidity REAL, pressure REAL,
    precip REAL, cloud_cover REAL, cape REAL, solar_rad REAL
);
SELECT create_hypertable('wrf_hindcast','time',if_not_exists=>TRUE);
CREATE INDEX idx_wrf_spot ON wrf_hindcast(spot_id,time DESC);

-- SWAN hindcast (oleaje)
CREATE TABLE IF NOT EXISTS swan_hindcast (
    time TIMESTAMPTZ NOT NULL, spot_id TEXT NOT NULL,
    lat REAL, lon REAL, wave_height REAL, wave_period REAL,
    wave_dir REAL, swell_height REAL, swell_period REAL, swell_dir REAL
);
SELECT create_hypertable('swan_hindcast','time',if_not_exists=>TRUE);
CREATE INDEX idx_swan_spot ON swan_hindcast(spot_id,time DESC);

-- ROMS hindcast (corrientes, SST)
CREATE TABLE IF NOT EXISTS roms_hindcast (
    time TIMESTAMPTZ NOT NULL, spot_id TEXT NOT NULL,
    lat REAL, lon REAL, current_u REAL, current_v REAL,
    sst REAL, salinity REAL
);
SELECT create_hypertable('roms_hindcast','time',if_not_exists=>TRUE);
CREATE INDEX idx_roms_spot ON roms_hindcast(spot_id,time DESC);

GRANT SELECT ON wrf_hindcast,swan_hindcast,roms_hindcast TO meteomap_app;

FASE 4: Verificar

-- Climatologia viento Cesantes julio 14-18h
SELECT EXTRACT(hour FROM time) AS hora,
    ROUND(AVG(wind_speed * 1.94384)::numeric, 1) AS avg_kt
FROM wrf_hindcast
WHERE spot_id = 'cesantes'
  AND EXTRACT(month FROM time) = 7
  AND EXTRACT(hour FROM time) BETWEEN 14 AND 18
GROUP BY hora ORDER BY hora;

-- Dias con SW 12-15kt (F5 "Dias como hoy")
SELECT DISTINCT DATE(time) AS dia,
    ROUND(AVG(wind_speed * 1.94384)::numeric, 1) AS avg_kt
FROM wrf_hindcast
WHERE spot_id = 'cesantes'
  AND wind_dir BETWEEN 200 AND 250
  AND wind_speed * 1.94384 BETWEEN 12 AND 15
  AND EXTRACT(hour FROM time) BETWEEN 12 AND 18
GROUP BY dia HAVING COUNT(*) >= 3
ORDER BY dia DESC LIMIT 20;
